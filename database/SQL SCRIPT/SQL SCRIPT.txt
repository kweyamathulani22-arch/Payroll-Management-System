-- =====================================================================
--  PAYROLL MANAGEMENT SYSTEM  -  MySQL 8.x schema for MySQL Workbench
--  This script creates an EMPTY system: no demo users, no employees,
--  no payroll. Every person who signs up starts with a clean company.
--
--  HOW TO RUN: open this file in MySQL Workbench (File > Open SQL Script)
--  and press the lightning-bolt "Execute" button (it runs the whole file).
--
--  WARNING: the DROP lines mean running this script again DELETES ALL
--  accounts and data. That is also how you wipe the database clean.
--  (Run it with an account that is allowed to CREATE USER, e.g. root.)
--
--  PRIVACY: every table has an owner_id (the user who owns the row).
--  The Java program only ever asks for rows where owner_id = the logged-in
--  user, and the foreign keys make it impossible to link a payroll record
--  to an employee that belongs to a different user.
-- =====================================================================

CREATE DATABASE IF NOT EXISTS payroll_management_system
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE payroll_management_system;

DROP VIEW  IF EXISTS v_monthly_summary;
DROP VIEW  IF EXISTS v_payslip_list;
DROP VIEW  IF EXISTS v_payroll_detail;
DROP TABLE IF EXISTS payslips;
DROP TABLE IF EXISTS payroll_deductions;
DROP TABLE IF EXISTS payroll_records;
DROP TABLE IF EXISTS employees;
DROP TABLE IF EXISTS departments;
DROP TABLE IF EXISTS company_settings;
DROP TABLE IF EXISTS users;

-- ---------------------------------------------------------------------
-- 1. users  (login.java / SignUp.java)
--    The password is never saved as text. We save a random "salt" and
--    password_hash = SHA-256(salt + password).
-- ---------------------------------------------------------------------
CREATE TABLE users (
    user_id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
    full_name      VARCHAR(100) NOT NULL,
    email          VARCHAR(120) NOT NULL,
    password_salt  CHAR(32)     NOT NULL,
    password_hash  CHAR(64)     NOT NULL,
    created_at     TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id),
    CONSTRAINT uq_users_email UNIQUE (email)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 2. company_settings  (Setting.java)
--    One row per user. These details are printed on that user's payslips.
-- ---------------------------------------------------------------------
CREATE TABLE company_settings (
    owner_id             INT UNSIGNED NOT NULL,
    company_name         VARCHAR(120) NOT NULL DEFAULT '',
    company_address      VARCHAR(255) NOT NULL DEFAULT '',
    currency             VARCHAR(60)  NOT NULL DEFAULT 'ZAR - South African rand (R)',
    standard_hours_week  DECIMAL(5,2) NOT NULL DEFAULT 40.00,
    overtime_multiplier  DECIMAL(4,2) NOT NULL DEFAULT 1.50,
    default_hours_month  DECIMAL(6,2) NOT NULL DEFAULT 160.00,
    PRIMARY KEY (owner_id),
    CONSTRAINT ck_settings_week  CHECK (standard_hours_week > 0 AND standard_hours_week <= 168),
    CONSTRAINT ck_settings_mult  CHECK (overtime_multiplier >= 1 AND overtime_multiplier <= 5),
    CONSTRAINT ck_settings_month CHECK (default_hours_month > 0 AND default_hours_month <= 744),
    CONSTRAINT fk_settings_owner FOREIGN KEY (owner_id) REFERENCES users (user_id)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 3. departments  (the "Department" box on AddEmployee.java)
--    Each user has their own list of departments.
-- ---------------------------------------------------------------------
CREATE TABLE departments (
    department_id  INT UNSIGNED NOT NULL AUTO_INCREMENT,
    owner_id       INT UNSIGNED NOT NULL,
    name           VARCHAR(80)  NOT NULL,
    PRIMARY KEY (department_id),
    CONSTRAINT uq_departments_owner_name UNIQUE (owner_id, name),
    CONSTRAINT uq_departments_id_owner   UNIQUE (department_id, owner_id),  -- lets employees point at "department + owner"
    CONSTRAINT fk_departments_owner FOREIGN KEY (owner_id) REFERENCES users (user_id)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 4. employees  (AddEmployee.java / employees.java)
--    The extra columns (gender, date of birth, job title, address, post code and
--    employment contract) come from the Add Employee screen.
--    is_active = 1 is 'Active', is_active = 0 is 'Not Active'.
--    employee_id is typed in by the user, so two different users may both
--    have an employee 1001. The key is therefore (owner_id, employee_id).
--    An employee is NEVER deleted: 'delete' only sets is_active = 0 ('Not Active'). The details and the
--    old payslips stay in the company database (a trigger below blocks DELETE, and the app user has no DELETE right).
-- ---------------------------------------------------------------------
CREATE TABLE employees (
    owner_id       INT UNSIGNED  NOT NULL,
    employee_id    INT UNSIGNED  NOT NULL,
    first_name     VARCHAR(50)   NOT NULL,
    last_name      VARCHAR(50)   NOT NULL,
    email          VARCHAR(120)  NOT NULL,
    phone          VARCHAR(20)   NOT NULL,
    gender         VARCHAR(10)   NOT NULL,
    date_of_birth  DATE          NOT NULL,
    job_title      VARCHAR(60)   NOT NULL,
    address        VARCHAR(150)  NOT NULL,
    post_code      CHAR(4)       NOT NULL,
    employment_contract VARCHAR(20) NOT NULL,           -- Permanent, Part-Time or Temporary
    department_id  INT UNSIGNED  NOT NULL,
    hourly_rate    DECIMAL(10,2) NOT NULL,
    is_active      TINYINT(1)    NOT NULL DEFAULT 1,
    created_at     TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (owner_id, employee_id),
    CONSTRAINT uq_employees_owner_email UNIQUE (owner_id, email),
    CONSTRAINT ck_employees_rate CHECK (hourly_rate > 0),
    CONSTRAINT ck_employees_gender CHECK (gender IN ('Male', 'Female', 'Other')),
    CONSTRAINT ck_employees_postcode CHECK (post_code REGEXP '^[0-9]{4}$'),
    CONSTRAINT ck_employees_contract CHECK (employment_contract IN ('Permanent', 'Part-Time', 'Temporary')),
    CONSTRAINT fk_employees_owner FOREIGN KEY (owner_id) REFERENCES users (user_id),
    CONSTRAINT fk_employees_department FOREIGN KEY (department_id, owner_id)
        REFERENCES departments (department_id, owner_id) ON UPDATE CASCADE ON DELETE RESTRICT,
    INDEX idx_employees_name (owner_id, last_name, first_name)
) ENGINE=InnoDB;

-- Safety net: nobody (not even the app by mistake) can delete an employee row.
-- To remove an employee from the lists, set is_active = 0 instead.
DELIMITER $$
CREATE TRIGGER trg_employees_never_deleted BEFORE DELETE ON employees
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Employees are never deleted. Set is_active = 0 (Not Active) instead.';
END$$
DELIMITER ;

-- ---------------------------------------------------------------------
-- 5. payroll_records  (PayrollProcessing.java, "Save Payroll")
--    One row per employee per month. The rate and multiplier used are
--    copied in, so a later rate change never alters an old payslip.
--    The CHECKs make the database refuse wrong arithmetic:
--      gross = normal + overtime + bonus,  net = gross - deductions,  net >= 0
--    and a record is either 'Pending' (no paid_at) or 'Paid' (has paid_at).
-- ---------------------------------------------------------------------
CREATE TABLE payroll_records (
    payroll_id           INT UNSIGNED  NOT NULL AUTO_INCREMENT,
    owner_id             INT UNSIGNED  NOT NULL,
    employee_id          INT UNSIGNED  NOT NULL,
    pay_year             SMALLINT UNSIGNED NOT NULL,
    pay_month            TINYINT UNSIGNED  NOT NULL,
    hourly_rate          DECIMAL(10,2) NOT NULL,
    normal_hours         DECIMAL(6,2)  NOT NULL,
    overtime_hours       DECIMAL(6,2)  NOT NULL DEFAULT 0.00,
    overtime_multiplier  DECIMAL(4,2)  NOT NULL DEFAULT 1.50,
    normal_pay           DECIMAL(12,2) NOT NULL,
    overtime_pay         DECIMAL(12,2) NOT NULL,
    bonus_pay            DECIMAL(12,2) NOT NULL DEFAULT 0.00,   -- "Bonuses & Incentives"
    bonus_description    VARCHAR(100)  NULL,                    -- what the bonus is for, e.g. Birthday gift (printed on the payslip)
    gross_pay            DECIMAL(12,2) NOT NULL,
    total_deductions     DECIMAL(12,2) NOT NULL DEFAULT 0.00,
    net_pay              DECIMAL(12,2) NOT NULL,
    payment_status       VARCHAR(10)   NOT NULL DEFAULT 'Pending',  -- 'Pending' until the Pay Employee button is used
    paid_at              TIMESTAMP     NULL DEFAULT NULL,
    created_at           TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (payroll_id),
    CONSTRAINT uq_payroll_period UNIQUE (owner_id, employee_id, pay_year, pay_month),
    CONSTRAINT uq_payroll_id_owner UNIQUE (payroll_id, owner_id),
    CONSTRAINT ck_payroll_month CHECK (pay_month BETWEEN 1 AND 12),
    CONSTRAINT ck_payroll_year  CHECK (pay_year BETWEEN 2000 AND 2100),
    CONSTRAINT ck_payroll_hours CHECK (normal_hours >= 0 AND overtime_hours >= 0
                                       AND normal_hours + overtime_hours <= 744),
    CONSTRAINT ck_payroll_mult  CHECK (overtime_multiplier >= 1),
    CONSTRAINT ck_payroll_bonus CHECK (bonus_pay >= 0),
    CONSTRAINT ck_payroll_gross CHECK (gross_pay = normal_pay + overtime_pay + bonus_pay),
    CONSTRAINT ck_payroll_net   CHECK (net_pay = gross_pay - total_deductions AND net_pay >= 0),
    CONSTRAINT ck_payroll_paid  CHECK ((payment_status = 'Paid' AND paid_at IS NOT NULL)
                                    OR (payment_status = 'Pending' AND paid_at IS NULL)),
    CONSTRAINT fk_payroll_employee FOREIGN KEY (owner_id, employee_id)
        REFERENCES employees (owner_id, employee_id) ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 6. payroll_deductions  ("Add deduction" on PayrollProcessing.java)
--    Any number of deductions per payroll record.
-- ---------------------------------------------------------------------
CREATE TABLE payroll_deductions (
    deduction_id  INT UNSIGNED  NOT NULL AUTO_INCREMENT,
    owner_id      INT UNSIGNED  NOT NULL,
    payroll_id    INT UNSIGNED  NOT NULL,
    description   VARCHAR(100)  NOT NULL,
    amount        DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (deduction_id),
    CONSTRAINT ck_deduction_amount CHECK (amount > 0),
    CONSTRAINT fk_deduction_payroll FOREIGN KEY (payroll_id, owner_id)
        REFERENCES payroll_records (payroll_id, owner_id) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 7. payslips  ("Generate Payslip" and payslip.java)
--    payslip_number counts 1, 2, 3 ... separately for each user, so one
--    user can never tell how many payslips another user has made.
-- ---------------------------------------------------------------------
CREATE TABLE payslips (
    payslip_id      INT UNSIGNED NOT NULL AUTO_INCREMENT,
    owner_id        INT UNSIGNED NOT NULL,
    payslip_number  INT UNSIGNED NOT NULL,
    payroll_id      INT UNSIGNED NOT NULL,
    generated_at    TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (payslip_id),
    CONSTRAINT uq_payslips_number  UNIQUE (owner_id, payslip_number),
    CONSTRAINT uq_payslips_payroll UNIQUE (payroll_id),
    CONSTRAINT fk_payslips_payroll FOREIGN KEY (payroll_id, owner_id)
        REFERENCES payroll_records (payroll_id, owner_id) ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- VIEWS (ready-made queries the screens use). Every view carries owner_id
-- and the Java code always adds  WHERE owner_id = <logged-in user>.
-- ---------------------------------------------------------------------

-- one row per payroll record with the employee, department and period text
CREATE VIEW v_payroll_detail AS
SELECT  p.payroll_id,
        p.owner_id,
        p.employee_id,
        CONCAT(e.first_name, ' ', e.last_name)  AS employee_name,
        d.name                                  AS department,
        p.pay_year,
        p.pay_month,
        CONCAT(ELT(p.pay_month, 'January','February','March','April','May','June',
                   'July','August','September','October','November','December'),
               ' ', p.pay_year)                 AS pay_period,
        p.hourly_rate,
        p.normal_hours,
        p.overtime_hours,
        p.overtime_multiplier,
        p.normal_pay,
        p.overtime_pay,
        p.bonus_pay,
        p.bonus_description,
        p.gross_pay,
        p.total_deductions,
        p.net_pay,
        p.payment_status,
        p.paid_at
FROM    payroll_records p
JOIN    employees   e ON e.owner_id = p.owner_id AND e.employee_id = p.employee_id
JOIN    departments d ON d.department_id = e.department_id AND d.owner_id = e.owner_id;

-- the columns of the table on payslip.java (Payslip No. | Employee | ID | Pay Period | Gross | Deductions)
CREATE VIEW v_payslip_list AS
SELECT  s.payslip_id,
        s.owner_id,
        s.payslip_number,
        CONCAT('PS-', LPAD(s.payslip_number, 6, '0')) AS payslip_label,
        v.employee_name,
        v.employee_id,
        v.pay_period,
        v.pay_year,
        v.pay_month,
        v.gross_pay,
        v.total_deductions,
        s.generated_at
FROM    payslips s
JOIN    v_payroll_detail v ON v.payroll_id = s.payroll_id AND v.owner_id = s.owner_id;

-- totals for each month (dashboard and report totals)
CREATE VIEW v_monthly_summary AS
SELECT  owner_id,
        pay_year,
        pay_month,
        MIN(pay_period)        AS pay_period,
        COUNT(*)               AS employees_paid,
        SUM(normal_hours)      AS normal_hours,
        SUM(overtime_hours)    AS overtime_hours,
        SUM(gross_pay)         AS gross_pay,
        SUM(total_deductions)  AS total_deductions,
        SUM(net_pay)           AS net_pay
FROM    v_payroll_detail
GROUP BY owner_id, pay_year, pay_month;

-- ---------------------------------------------------------------------
-- APPLICATION USER: the Java program connects with this account (not root).
-- If you change the password here, change it in db.properties too.
-- ---------------------------------------------------------------------
CREATE USER IF NOT EXISTS 'payroll_app'@'localhost' IDENTIFIED BY 'Payroll@2026';
REVOKE ALL PRIVILEGES, GRANT OPTION FROM 'payroll_app'@'localhost';   -- start from nothing (also cleans old grants)
GRANT SELECT, INSERT         ON payroll_management_system.users              TO 'payroll_app'@'localhost';
GRANT SELECT, INSERT, UPDATE ON payroll_management_system.company_settings   TO 'payroll_app'@'localhost';
GRANT SELECT, INSERT, UPDATE ON payroll_management_system.departments        TO 'payroll_app'@'localhost';
GRANT SELECT, INSERT, UPDATE ON payroll_management_system.employees          TO 'payroll_app'@'localhost';   -- no DELETE: employees stay in the database
GRANT SELECT, INSERT, UPDATE, DELETE ON payroll_management_system.payroll_records TO 'payroll_app'@'localhost'; -- a Pending payroll can be replaced
GRANT SELECT, INSERT         ON payroll_management_system.payroll_deductions TO 'payroll_app'@'localhost';
GRANT SELECT, INSERT, UPDATE ON payroll_management_system.payslips           TO 'payroll_app'@'localhost';
GRANT SELECT ON payroll_management_system.v_payroll_detail  TO 'payroll_app'@'localhost';
GRANT SELECT ON payroll_management_system.v_payslip_list    TO 'payroll_app'@'localhost';
GRANT SELECT ON payroll_management_system.v_monthly_summary TO 'payroll_app'@'localhost';
FLUSH PRIVILEGES;

-- (end of script - the database is empty, create your first account with "Sign Up") and enjoy the Payroll System Management
